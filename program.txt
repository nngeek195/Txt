@echo off
setlocal

:: ============================================================
:: Home Admin Agent - Windows SSH Installer (fixed)
:: Port: 3014
:: Run this BAT file as Administrator
:: ============================================================

echo.
echo ============================================
echo       HOME ADMIN - INSTALLER
echo ============================================
echo.

:: ------------------------------------------------------------
:: Check Administrator privileges
:: ------------------------------------------------------------
net session >nul 2>&1

if %errorlevel% neq 0 (
    echo [ERROR] Please run this file as Administrator.
    echo.
    pause
    exit /b 1
)

echo [OK] Administrator privileges detected.
echo.

:: ------------------------------------------------------------
:: Configuration
:: ------------------------------------------------------------
set "ADMIN_USER=HomeAdmin"
set "SSH_PORT=3014"

:: ------------------------------------------------------------
:: Install OpenSSH Server
:: ------------------------------------------------------------
echo [1/6] Checking OpenSSH Server...

powershell -NoProfile -ExecutionPolicy Bypass -Command ^
    "$ErrorActionPreference='Stop'; try { $cap = Get-WindowsCapability -Online | Where-Object { $_.Name -like 'OpenSSH.Server*' } | Select-Object -First 1; if (-not $cap) { Write-Host '[ERROR] Could not find OpenSSH.Server capability on this image.'; exit 1 }; if ($cap.State -ne 'Installed') { Add-WindowsCapability -Online -Name $cap.Name | Out-Null }; exit 0 } catch { Write-Host \"[ERROR] $($_.Exception.Message)\"; exit 1 }"

if %errorlevel% neq 0 (
    echo [ERROR] Failed to install OpenSSH Server. Check your connection to Windows Update.
    pause
    exit /b 1
)

echo [OK] OpenSSH Server installed.
echo.

:: ------------------------------------------------------------
:: Ensure default sshd_config exists
:: (Windows only generates it the first time the service starts)
:: ------------------------------------------------------------
echo [2/6] Ensuring default SSH config exists...

powershell -NoProfile -ExecutionPolicy Bypass -Command ^
    "$ErrorActionPreference='Stop'; try { $cfg='C:\ProgramData\ssh\sshd_config'; if (-not (Test-Path $cfg)) { Set-Service -Name sshd -StartupType Manual -ErrorAction SilentlyContinue; Start-Service sshd; Start-Sleep -Seconds 2; Stop-Service sshd }; if (-not (Test-Path $cfg)) { Write-Host '[ERROR] sshd_config still not present after starting service once.'; exit 1 }; exit 0 } catch { Write-Host \"[ERROR] $($_.Exception.Message)\"; exit 1 }"

if %errorlevel% neq 0 (
    echo [ERROR] Could not generate default sshd_config.
    pause
    exit /b 1
)

echo [OK] SSH config file present.
echo.

:: ------------------------------------------------------------
:: Create/configure administrator account
:: ------------------------------------------------------------
echo [3/6] Creating/configuring administrator account...
echo.
echo Enter the password for %ADMIN_USER%.
echo The password will NOT be stored in this BAT file.
echo.

powershell -NoProfile -ExecutionPolicy Bypass -Command ^
    "$ErrorActionPreference='Stop'; try { $p = Read-Host 'Password' -AsSecureString; $existing = Get-LocalUser -Name '%ADMIN_USER%' -ErrorAction SilentlyContinue; if (-not $existing) { New-LocalUser -Name '%ADMIN_USER%' -Password $p -Description 'Home network remote administration account' | Out-Null } else { Set-LocalUser -Name '%ADMIN_USER%' -Password $p }; $inGroup = (Get-LocalGroupMember -Group 'Administrators' -ErrorAction SilentlyContinue | Where-Object { $_.Name -like \"*$env:COMPUTERNAME\%ADMIN_USER%\" -or $_.Name -like \"*\%ADMIN_USER%\" }); if (-not $inGroup) { Add-LocalGroupMember -Group 'Administrators' -Member '%ADMIN_USER%' }; exit 0 } catch { Write-Host \"[ERROR] $($_.Exception.Message)\"; exit 1 }"

if %errorlevel% neq 0 (
    echo [ERROR] Failed to create/configure the account.
    pause
    exit /b 1
)

echo [OK] Administrator account configured.
echo.

:: ------------------------------------------------------------
:: Configure SSH (port + password auth)
:: ------------------------------------------------------------
echo [4/6] Configuring SSH...

powershell -NoProfile -ExecutionPolicy Bypass -Command ^
    "$ErrorActionPreference='Stop'; try { $cfg='C:\ProgramData\ssh\sshd_config'; Copy-Item $cfg ($cfg+'.backup') -Force; $lines = Get-Content $cfg | Where-Object { $_ -notmatch '^\s*#?\s*Port\s+' -and $_ -notmatch '^\s*#?\s*PasswordAuthentication\s+' }; $lines += 'Port 3014'; $lines += 'PasswordAuthentication yes'; Set-Content -Path $cfg -Value $lines -Encoding ascii; exit 0 } catch { Write-Host \"[ERROR] $($_.Exception.Message)\"; exit 1 }"

if %errorlevel% neq 0 (
    echo [ERROR] Failed to configure SSH.
    pause
    exit /b 1
)

echo [OK] SSH configured for port 3014.
echo.

:: ------------------------------------------------------------
:: Configure firewall
:: ------------------------------------------------------------
echo [5/6] Configuring Windows Firewall...

powershell -NoProfile -ExecutionPolicy Bypass -Command ^
    "$ErrorActionPreference='Stop'; try { Remove-NetFirewallRule -DisplayName 'HomeAdmin SSH 3014' -ErrorAction SilentlyContinue; New-NetFirewallRule -DisplayName 'HomeAdmin SSH 3014' -Direction Inbound -Protocol TCP -LocalPort 3014 -Action Allow -Profile Private,Domain | Out-Null; $profile = (Get-NetConnectionProfile | Select-Object -First 1).NetworkCategory; if ($profile -eq 'Public') { Write-Host '[WARN] Your active network profile is set to Public. The firewall rule will NOT apply until you set it to Private.' }; exit 0 } catch { Write-Host \"[ERROR] $($_.Exception.Message)\"; exit 1 }"

if %errorlevel% neq 0 (
    echo [ERROR] Failed to configure firewall.
    pause
    exit /b 1
)

echo [OK] Firewall rule created for Private/Domain networks.
echo.

:: ------------------------------------------------------------
:: Configure and start SSH service
:: ------------------------------------------------------------
echo [6/6] Starting SSH service...

powershell -NoProfile -ExecutionPolicy Bypass -Command ^
    "$ErrorActionPreference='Stop'; try { Set-Service -Name sshd -StartupType Automatic; Restart-Service sshd; Start-Sleep -Seconds 1; $svc = Get-Service sshd; if ($svc.Status -ne 'Running') { Write-Host '[ERROR] sshd did not reach Running state.'; exit 1 }; exit 0 } catch { Write-Host \"[ERROR] $($_.Exception.Message)\"; exit 1 }"

if %errorlevel% neq 0 (
    echo [ERROR] Failed to start SSH service.
    pause
    exit /b 1
)

echo [OK] SSH service is running.
echo.

:: ------------------------------------------------------------
:: Display machine information
:: ------------------------------------------------------------
echo Machine info:
echo.

hostname

echo.
echo Network addresses:
powershell -NoProfile -Command ^
    "Get-NetIPAddress -AddressFamily IPv4 | Where-Object {$_.IPAddress -notlike '127.*' -and $_.IPAddress -notlike '169.254.*'} | Select-Object IPAddress,InterfaceAlias"

echo.
echo Active network profile:
powershell -NoProfile -Command ^
    "Get-NetConnectionProfile | Select-Object InterfaceAlias,NetworkCategory"

echo.
echo ============================================
echo          INSTALLATION COMPLETE
echo ============================================
echo.
echo Username : %ADMIN_USER%
echo SSH Port : %SSH_PORT%
echo.
echo Connect from another PC using:
echo.
echo ssh -p 3014 %ADMIN_USER%@MACHINE_IP
echo.
echo Example:
echo ssh -p 3014 HomeAdmin@192.168.23.32
echo.
echo IMPORTANT:
echo - Do NOT expose port 3014 to the Internet (no router port-forwarding).
echo - This only works if the active network profile above says "Private" (or "Domain").
echo - If it says "Public", run: Set-NetConnectionProfile -NetworkCategory Private
echo   (only do this on your trusted home network).
echo.

pause
endlocal
